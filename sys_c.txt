#include <iostream>
using namespace std;

#define INF  999

void diji (int grid[10][10] , int src , int n){
    int dis[10] , vis[10] = {0};

    for(int i =0 ; i< n ; i++) {
        dis[i] = INF;
    }

    dis[src] = 0;
    for(int j = 0; j<n-1; j++){
        int min = INF;
        int u = -1;
        for(int i =0 ; i< n; i++){
            if(vis[i] != 1 && dis[i] < min){
                min = dis[i];
                u =i;
            }
        }
        vis[u]= 1;
        for(int v =0; v<n; v++){
            if(vis[v] != 1 && grid[u][v] != 0 && dis[u] + grid[u][v] < dis[v]){
                dis[v] = dis[u] + grid[u][v];
            }
        }
    }
    cout<<"distance from source to node\n";
    for(int i =0 ; i<n; i++){
        cout<<"source:"<<src<<"-> Node :"<<i<<" = "<<dis[i]<<endl;
    }
}

#include <iostream>
using namespace std;
const string gen = "10011";

string crc(string data){
    int n = gen.length();
    string temp = data + string (n-1, '0');
    for(int i =0 ; i<data.length(); i++){
        if(temp[i]=='1'){
            for(int j =0 ; j<n ; j++){
                temp[i+j] = (temp[i+j] == gen[j]) ? '0':'1';
            }
        }
    }
    return temp.substr(temp.length() );
}

#include<iostream>
#include<unistd.h>
#include<sys/socket.h>
#include<arpa/inet.h>
using namespace std;

int main(){
    int s=socket(AF_INET,SOCK_STREAM,0);
    sockaddr_in a{};
    a.sin_family=AF_INET;
    a.sin_port=htons(8080);
    a.sin_addr.s_addr=INADDR_ANY;

    bind(s,(sockaddr*)&a,sizeof(a));
    listen(s,1);

    cout<<"Waiting...\n";
    int c=accept(s,NULL,NULL);

    int n,w,lost;
    recv(c,&n,sizeof(n),MSG_WAITALL);
    recv(c,&w,sizeof(w),MSG_WAITALL);
    recv(c,&lost,sizeof(lost),MSG_WAITALL);

    int expected=1, dropped=0;

    while(expected<=n){
        int count;
        recv(c,&count,4,MSG_WAITALL);

        for(int i=0;i<count;i++){
            int f;
            recv(c,&f,4,MSG_WAITALL);

            if(f==lost && !dropped){
                cout<<"Frame "<<f<<" lost\n";
                dropped=1;
            }
            else if(f==expected){
                cout<<"Frame "<<f<<" received\n";
                expected++;
            }
            else
                cout<<"Frame "<<f<<" discarded\n";
        }

        int ack=expected-1;
        cout<<"ACK "<<ack<<endl;
        send(c,&ack,4,0);
    }

    close(c);
    close(s);
    return 0;
}

#include<iostream>
#include<unistd.h>
#include<sys/socket.h>
#include<arpa/inet.h>
using namespace std;

int main(){
    int c=socket(AF_INET,SOCK_STREAM,0);
    sockaddr_in a{};
    a.sin_family=AF_INET;
    a.sin_port=htons(8080);
    a.sin_addr.s_addr=inet_addr("127.0.0.1");

    connect(c,(sockaddr*)&a,sizeof(a));

    int n,w,lost;
    cout<<"Enter frames, window size and lost frame: ";
    cin>>n>>w>>lost;

    send(c,&n,4,0);
    send(c,&w,4,0);
    send(c,&lost,4,0);

    int base=1;

    while(base<=n){
        int count=min(w,n-base+1);
        send(c,&count,4,0);

        for(int i=base;i<base+count;i++){
            cout<<"Sending frame "<<i<<endl;
            send(c,&i,4,0);
        }

        int ack;
        recv(c,&ack,4,MSG_WAITALL);
        cout<<"ACK received: "<<ack<<endl;

        base=ack+1;
    }

    cout<<"All frames sent\n";
    close(c);
    return 0;
}

9 leaky bucket

#include <iostream>
using namespace std;

int main() {
    int bucket_size, output_rate, n;
    int bucket = 0;

    cout << "Enter bucket size: ";
    cin >> bucket_size;

    cout << "Enter output rate: ";
    cin >> output_rate;

    cout << "Enter number of inputs: ";
    cin >> n;

    for (int i = 1; i <= n; i++) {
        int incoming;

        cout << "\nEnter incoming packets: ";
        cin >> incoming;

        if (bucket + incoming > bucket_size) {
            cout << "Bucket overflow! "
                 << bucket + incoming - bucket_size
                 << " packets discarded.\n";

            bucket = bucket_size;
        } else {
            bucket += incoming;
        }

        int sent = min(bucket, output_rate);
        bucket -= sent;

        cout << "Packets sent: " << sent << endl;
        cout << "Packets remaining: " << bucket << endl;
    }

    // Send remaining packets
    while (bucket > 0) {
        int sent = min(bucket, output_rate);
        bucket -= sent;

        cout << "\nPackets sent: " << sent << endl;
        cout << "Packets remaining: " << bucket << endl;
    }

    return 0;
}

# NS2 Lab 5: Two Ethernet LANs, UDP traffic, error model and throughput
set ns [new Simulator]
set tf [open lab5.tr w]
$ns trace-all $tf
set nf [open lab5.nam w]
$ns namtrace-all $nf
$ns color 1 red

for {set i 0} {$i < 7} {incr i} {
    set n($i) [$ns node]
}
$n(1) label "UDP Source"
$n(3) label "LAN1 Gateway"
$n(5) label "Destination"
$ns make-lan "$n(0) $n(1) $n(2) $n(3)" 10Mb 10ms LL Queue/DropTail Mac/802_3
$ns make-lan "$n(4) $n(5) $n(6)" 10Mb 10ms LL Queue/DropTail Mac/802_3
$ns duplex-link $n(3) $n(6) 1Mb 10ms DropTail

set udp [new Agent/UDP]
$udp set class_ 1
$ns attach-agent $n(1) $udp
set sink [new Agent/Null]
$ns attach-agent $n(5) $sink
$ns connect $udp $sink

set cbr [new Application/Traffic/CBR]
$cbr attach-agent $udp
$cbr set packetSize_ 1000
$cbr set interval_ 0.001

set err [new ErrorModel]
$err set rate_ 0.2
$err unit pkt
$ns lossmodel $err $n(3) $n(6)

proc finish {} {
    global ns tf nf
    $ns flush-trace
    close $tf
    close $nf
    catch {exec nam lab5.nam &}
    exit 0
}
$ns at 0.1 "$cbr start"
$ns at 4.8 "$cbr stop"
$ns at 5.0 "finish"
$ns run

BEGIN { bytes = 0; first = -1; last = 0 }
$1 == "r" && $4 == "5" && $5 == "cbr" {
    if (first < 0) first = $2
    last = $2
    bytes += $6
}
END {
    duration = last - first
    if (duration > 0)
        printf("Throughput: %.6f Mbps\n", bytes * 8 / duration / 1000000)
    else
        print "Throughput unavailable: no received CBR packets were found."
}
