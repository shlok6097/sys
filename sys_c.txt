void diji (int grid[10][10] , int src , int n){
    int dis[10] , vis[10] = {0};

    for(int i =0 ; i< n ; i++) {
        dis[i] = INF;
    }

    dis[src] = 0;
    for(int j = 0; j<n-1; j++){
        int min = INF;
        int u = -1;
        for(int i =0 ; i< n; i++){
            if(vis[i] != 1 && dis[i] < min){
                min = dis[i];
                u =i;
            }
        }
        vis[u]= 1;
        for(int v =0; v<n; v++){
            if(vis[v] != 1 && grid[u][v] != 0 && dis[u] + grid[u][v] < dis[v]){
                dis[v] = dis[u] + grid[u][v];
            }
        }
    }
    cout<<"distance from source to node\n";
    for(int i =0 ; i<n; i++){
        cout<<"source:"<<src<<"-> Node :"<<i<<" = "<<dis[i]<<endl;
    }
}

#include <iostream>

const string gen = "10011";

string crc(string data){
    int n = gen.length();
    string temp = data + string (n-1, '0');
    for(int i =0 ; i<data.length()-n; i++){
        if(temp[i]=='1'){
            for(int j =0 ; j<n ; j++){
                temp[i+j] = (temp[i+j] == gen[j]) ? '0':'1';
            }
        }
    }
    return temp.substr(temp.length() - n-1);
}

#include <iostream>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

using namespace std;

int main() {
    int s = socket(AF_INET, SOCK_STREAM, 0);

    sockaddr_in a{};
    a.sin_family = AF_INET;
    a.sin_port = htons(8080);
    a.sin_addr.s_addr = INADDR_ANY;

    bind(s, (sockaddr*)&a, sizeof(a));
    listen(s, 1);

    cout << "Waiting for sender...\n";
    int c = accept(s, NULL, NULL);

    int lost, n, w;

    cout << "Enter frame number to simulate loss (-1 for none): ";
    cin >> lost;

    recv(c, &n, sizeof(n), MSG_WAITALL);
    recv(c, &w, sizeof(w), MSG_WAITALL);

    int expected = 0;

    while (expected < n) {
        int frame;

        if (recv(c, &frame, sizeof(frame), MSG_WAITALL) <= 0)
            break;

        if (frame == lost) {
            cout << "Frame " << frame << " lost!\n";
            lost = -1;

            int ack = expected - 1;
            send(c, &ack, sizeof(ack), 0);
            cout << "Duplicate ACK " << ack << " sent\n";
            continue;
        }

        if (frame == expected) {
            cout << "Frame " << frame
                 << " received correctly\n";

            int ack = expected;
            expected++;

            send(c, &ack, sizeof(ack), 0);
            cout << "ACK " << ack << " sent\n";
        } else {
            cout << "Frame " << frame
                 << " discarded (out of order)\n";

            int ack = expected - 1;
            send(c, &ack, sizeof(ack), 0);
            cout << "Duplicate ACK " << ack << " sent\n";
        }
    }

    close(c);
    close(s);
    return 0;
}

gbn_client


#include <iostream>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

using namespace std;

int main() {
    int s = socket(AF_INET, SOCK_STREAM, 0);

    sockaddr_in a{};
    a.sin_family = AF_INET;
    a.sin_port = htons(8080);
    a.sin_addr.s_addr = inet_addr("127.0.0.1");

    if (connect(s, (sockaddr*)&a, sizeof(a)) < 0) {
        cout << "Connection failed\n";
        close(s);
        return 1;
    }

    int n, w;

    cout << "Enter number of frames: ";
    cin >> n;

    cout << "Enter window size: ";
    cin >> w;

    if (n <= 0 || w <= 0) {
        cout << "Invalid input\n";
        close(s);
        return 1;
    }

    send(s, &n, sizeof(n), 0);
    send(s, &w, sizeof(w), 0);

    int base = 0;

    while (base < n) {
        int end = min(base + w, n);

        // Send all frames in the current window
        for (int i = base; i < end; i++) {
            send(s, &i, sizeof(i), 0);
            cout << "Frame " << i << " sent\n";
        }

        // Receive ACK responses for this window
        int highestAck = base - 1;

        for (int i = base; i < end; i++) {
            int ack;

            if (recv(s, &ack, sizeof(ack), MSG_WAITALL) <= 0) {
                cout << "Connection closed\n";
                close(s);
                return 1;
            }

            cout << "ACK " << ack << " received\n";

            if (ack > highestAck)
                highestAck = ack;
        }

        // Advance the window using cumulative ACKs
        if (highestAck >= base) {
            base = highestAck + 1;
        }

        // Go back and retransmit unacknowledged frames
        while (base < end) {
            cout << "\nGo-Back-N: Retransmitting from Frame "
                 << base << endl;

            int retransmitEnd = end;

            for (int i = base; i < retransmitEnd; i++) {
                send(s, &i, sizeof(i), 0);
                cout << "Frame " << i << " retransmitted\n";
            }

            highestAck = base - 1;

            for (int i = base; i < retransmitEnd; i++) {
                int ack;

                if (recv(s, &ack, sizeof(ack), MSG_WAITALL) <= 0) {
                    close(s);
                    return 1;
                }

                cout << "ACK " << ack << " received\n";

                if (ack > highestAck)
                    highestAck = ack;
            }

            if (highestAck >= base)
                base = highestAck + 1;
        }
    }

    cout << "\nAll frames transmitted successfully\n";

    close(s);
    return 0;
}

#include <iostream>

#include <sys/stat.h>
#include <unistd.h>
#include <cstring>

using namespace std;

#define FIFO_NAME "myfifo"

int main()
{
    char buffer[1024];

    // Create FIFO
    mkfifo(FIFO_NAME, 0666);

    // Open file to be transferred
    int file = open("input.txt", O_RDONLY);

    if (file == -1)
    {
        perror("Error opening input.txt");
        return 1;
    }

    cout << "Server: Waiting for client..." << endl;

    // Open FIFO for writing
    int fifo = open(FIFO_NAME, O_WRONLY);

    cout << "Server: Sending file..." << endl;

    int bytes;

    // Read file and send through FIFO
    while ((bytes = read(file, buffer, sizeof(buffer))) > 0)
    {
        write(fifo, buffer, bytes);
    }

    cout << "Server: File transferred successfully." << endl;

    close(file);
    close(fifo);

    return 0;
}

    server.sin_family = AF_INET;

    ofstream file("received.txt", ios::binary);

    cout << "Server waiting for file..." << endl;

    while (true)
    {
        int n = recvfrom(sockfd, buffer, sizeof(buffer), 0,

            continue;

#include <iostream>
#include <fstream>
#include <cstring>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

using namespace std;

int main()
{
    int sockfd;
    char buffer[1024];

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    sockaddr_in server{};
    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    server.sin_addr.s_addr = inet_addr("127.0.0.1");

    ifstream file("input.txt", ios::binary);

    if (!file)
    {
        cout << "Cannot open input.txt" << endl;
        close(sockfd);
        return 1;
    }

    cout << "Sending file..." << endl;

    while (file.read(buffer, sizeof(buffer)) || file.gcount() > 0)
    {
        sendto(sockfd, buffer, file.gcount(), 0,
               (sockaddr*)&server, sizeof(server));
    }

    sendto(sockfd, "END", 3, 0,
           (sockaddr*)&server, sizeof(server));

    cout << "File sent successfully!" << endl;

    file.close();
    close(sockfd);
    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int bucket_size, output_rate, n;
    int bucket = 0;

    cout << "Enter bucket size: ";
    cin >> bucket_size;

    cout << "Enter output rate: ";
    cin >> output_rate;

    cout << "Enter number of inputs: ";
    cin >> n;

    for (int i = 1; i <= n; i++) {
        int incoming;

        cout << "\nEnter incoming packets: ";
        cin >> incoming;

        if (bucket + incoming > bucket_size) {
            cout << "Bucket overflow! "
                 << bucket + incoming - bucket_size
                 << " packets discarded.\n";

            bucket = bucket_size;
        } else {
            bucket += incoming;
        }

        int sent = min(bucket, output_rate);
        bucket -= sent;

        cout << "Packets sent: " << sent << endl;
        cout << "Packets remaining: " << bucket << endl;
    }

    // Send remaining packets
    while (bucket > 0) {
        int sent = min(bucket, output_rate);
        bucket -= sent;

        cout << "\nPackets sent: " << sent << endl;
        cout << "Packets remaining: " << bucket << endl;
    }

    return 0;
}

# Classic NS2 trace: sum received packet bytes at destination node 5.
BEGIN { bytes = 0; first = -1; last = 0 }
$1 == "r" && $4 == "5" && $5 == "cbr" {
    if (first < 0) first = $2
    last = $2
    bytes += $6
}
END {
    duration = last - first
    if (duration > 0)
        printf("Throughput: %.6f Mbps\n", bytes * 8 / duration / 1000000)
    else
        print "Throughput unavailable: no received CBR packets were found."
}
