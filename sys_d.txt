
CREATE DATABASE LibraryDB;
USE LibraryDB;

-- 1. Publisher details
CREATE TABLE PUBLISHER (
    Pub_id INT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Address VARCHAR(200),
    Phone VARCHAR(15)
);

-- 2. Book details
CREATE TABLE BOOK (
    Book_id INT PRIMARY KEY,
    Title VARCHAR(150) NOT NULL,
    Pub_id INT,
    Pub_Year YEAR,
    FOREIGN KEY (Pub_id)
        REFERENCES PUBLISHER(Pub_id)
        ON UPDATE CASCADE
        ON DELETE SET NULL
);

-- 3. Authors of books
CREATE TABLE BOOK_AUTHORS (
    Book_id INT,
    Author_Name VARCHAR(100),
    PRIMARY KEY (Book_id, Author_Name),
    FOREIGN KEY (Book_id)
        REFERENCES BOOK(Book_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

-- 4. Library branches
CREATE TABLE LIBRARY_BRANCH (
    Branch_id INT PRIMARY KEY,
    Branch_Name VARCHAR(100) NOT NULL,
    Address VARCHAR(200)
);

-- 5. Copies available at each branch
CREATE TABLE BOOK_COPIES (
    Book_id INT,
    Branch_id INT,
    No_of_Copies INT NOT NULL DEFAULT 0,
    PRIMARY KEY (Book_id, Branch_id),
    FOREIGN KEY (Book_id)
        REFERENCES BOOK(Book_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    FOREIGN KEY (Branch_id)
        REFERENCES LIBRARY_BRANCH(Branch_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    CHECK (No_of_Copies >= 0)
);

-- 6. Book lending transactions
CREATE TABLE BOOK_LENDING (
    Book_id INT,
    Branch_id INT,
    Card_No INT,
    Date_Out DATE,
    Due_Date DATE,
    PRIMARY KEY (Book_id, Branch_id, Card_No, Date_Out),
    FOREIGN KEY (Book_id, Branch_id)
        REFERENCES BOOK_COPIES(Book_id, Branch_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

SELECT
    B.Book_id,
    B.Title,
    P.Name AS Publisher_Name,
    A.Author_Name,
    LB.Branch_Name,
    BC.No_of_Copies
FROM BOOK B
JOIN PUBLISHER P
    ON B.Pub_id = P.Pub_id
LEFT JOIN BOOK_AUTHORS A
    ON B.Book_id = A.Book_id
LEFT JOIN BOOK_COPIES BC
    ON B.Book_id = BC.Book_id
LEFT JOIN LIBRARY_BRANCH LB
    ON BC.Branch_id = LB.Branch_id
ORDER BY B.Book_id, LB.Branch_id, A.Author_Name;

       'Has customers in city' AS Status
FROM SALESMAN S
WHERE EXISTS (
    SELECT 1
    FROM CUSTOMER C
    WHERE C.City = S.City
)

UNION

SELECT S.Name, S.City,
       'No customers in city' AS Status
FROM SALESMAN S
WHERE NOT EXISTS (
    SELECT 1
    FROM CUSTOMER C
    WHERE C.City = S.City
);


Q4.

CREATE VIEW Highest_Daily_Order AS
SELECT
    S.Salesman_id,
    S.Name AS Salesman_Name,
    C.Cust_Name,
    O.Ord_Date,
    O.Purchase_Amt
FROM SALESMAN S
JOIN ORDERS O
    ON S.Salesman_id = O.Salesman_id
JOIN CUSTOMER C
    ON O.Customer_id = C.Customer_id
WHERE O.Purchase_Amt = (
    SELECT MAX(O2.Purchase_Amt)
    FROM ORDERS O2
    WHERE O2.Ord_Date = O.Ord_Date
);

    Mov_id INT PRIMARY KEY,
    Rev_Stars INT,
    FOREIGN KEY (Mov_id, Rev_Stars) REFERENCES MOVIES(Mov_id)

CREATE DATABASE CollegeDB;
USE CollegeDB;

CREATE TABLE STUDENT (
    USN VARCHAR(15) PRIMARY KEY,
    SName VARCHAR(50) NOT NULL,
    Address VARCHAR(100),
    Phone VARCHAR(15),
    Gender CHAR(1)
);

CREATE TABLE SEMSEC (
    SSID INT PRIMARY KEY,
    Sem INT NOT NULL,
    Sec CHAR(1) NOT NULL,
    UNIQUE (Sem, Sec)
);

CREATE TABLE CLASS (
    USN VARCHAR(15),
    SSID INT,
    PRIMARY KEY (USN, SSID),
    FOREIGN KEY (USN) REFERENCES STUDENT(USN)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (SSID) REFERENCES SEMSEC(SSID)
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE SUBJECT (
    Subcode VARCHAR(10) PRIMARY KEY,
    Title VARCHAR(50) NOT NULL,
    Sem INT NOT NULL,
    Credits INT NOT NULL
);

CREATE TABLE IA_MARKS (
    USN VARCHAR(15),
    Subcode VARCHAR(10),
    SSID INT,
    Test1 DECIMAL(5,2),
    Test2 DECIMAL(5,2),
    Test3 DECIMAL(5,2),
    FinalIA DECIMAL(5,2),
    PRIMARY KEY (USN, Subcode, SSID),
    FOREIGN KEY (USN) REFERENCES STUDENT(USN)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (Subcode) REFERENCES SUBJECT(Subcode)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (SSID) REFERENCES SEMSEC(SSID)
        ON DELETE CASCADE ON UPDATE CASCADE
);

    	S.USN,
    	S.SName,
    	SS.Sem,
    	SS.Sec,
    	I.Subcode,
    	I.FinalIA,
	CASE
        	WHEN I.FinalIA BETWEEN 17 AND 20 THEN 'Outstanding'
       		WHEN I.FinalIA BETWEEN 12 AND 16 THEN 'Average'
        	WHEN I.FinalIA < 12 THEN 'Weak'
	END AS CAT

	ON I.USN = S.USN
	AND I.SSID = SS.SSID
WHERE SS.Sem = 8
 	AND SS.Sec IN ('A', 'B', 'C');

CREATE DATABASE CompanyDB;

CREATE TABLE EMPLOYEE (
    SSN       VARCHAR(10) PRIMARY KEY,
    Name      VARCHAR(50) NOT NULL,
    Address   VARCHAR(100),
    Sex       CHAR(1),
    Salary    DECIMAL(12,2),
    SuperSSN  VARCHAR(10),
    DNo       INT,

CREATE TABLE DEPARTMENT (
    DNo          INT PRIMARY KEY,
    DName        VARCHAR(50) NOT NULL UNIQUE,
    MgrSSN       VARCHAR(10),

    DNo   INT,

    PNo        INT PRIMARY KEY,
    PName      VARCHAR(50) NOT NULL,
    PLocation  VARCHAR(50),

SELECT P.PNo
FROM PROJECT P
JOIN WORKS_ON W ON W.PNo = P.PNo
JOIN EMPLOYEE E ON E.SSN = W.SSN
WHERE E.Name LIKE '% Scott'

UNION

SELECT P.PNo
FROM PROJECT P
JOIN DEPARTMENT D ON D.DNo = P.DNo
JOIN EMPLOYEE E ON E.SSN = D.MgrSSN
WHERE E.Name LIKE '% Scott';

Q2.

SELECT E.Name, P.PName, E.Salary * 1.10 AS New_Sal
FROM EMPLOYEE E
JOIN WORKS_ON W
	ON W.SSN = E.SSN
JOIN PROJECT P
	ON P.PNo = W.PNo
WHERE P.PName = 'IoT';

Q3.

SELECT 
	MAX(E.Salary) AS Max_sal, 
	MIN(E.Salary) AS min_sal, 
	AVG(E.Salary) AS avg_sal 
FROM EMPLOYEE E
JOIN DEPARTMENT D 
	ON D.DNo = E.DNo
WHERE D.DName = 'Accounts';

Q4.

SELECT E.Name
FROM EMPLOYEE E
WHERE NOT EXISTS (
    SELECT 1
    FROM PROJECT P
    WHERE P.DNo = 5
      AND NOT EXISTS (
          SELECT 1
          FROM WORKS_ON W
          WHERE W.SSN = E.SSN
            AND W.PNo = P.PNo
      )
);

Q5.

SELECT
    DNo,
    COUNT(CASE
        WHEN Salary > 600000 THEN 1
    END) AS High_Salary_Employees
FROM EMPLOYEE
GROUP BY DNo
HAVING COUNT(SSN) > 3;
