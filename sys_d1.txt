select s.name as salesman_name , c.name,s.city as same_city
from salesman s
join customer c on c.salesman_id = s.salesman_id
where s.city = c.city

union


select s.name as salesman_name , c.name,s.city as not_same_city
from salesman s
join customer c on c.salesman_id = s.salesman_id
where s.city != c.city ;

4.2 

select ss.sem , ss.sec ,
sum(case when s.gender = 'm' then 1 else 0 end) as male,
sum(case when s.gender = 'f' then 1 else 0 end) as female
from student s
join class c on c.usn = s.usn
join semsec ss on ss.ssid = c.ssid
group by ss.sem , ss.sec
order by ss.sem , ss.sec;
